```python
    # --- 五端渐进规则资产(Change A 产物;设计文档 §6.6.2) ---
    import re as _re
    rules_dir = root / ".claude" / "rules"
    fm_ok, fm_detail = True, ""
    for p in sorted(rules_dir.glob("*.md")) if rules_dir.is_dir() else []:
        if p.name == "README.md":
            continue
        m = _re.match(r"^---\n(.*?\n)---\n", p.read_text(encoding="utf-8"), _re.S)
        if not m or "description:" not in m.group(1):
            fm_ok, fm_detail = False, f"{p.name} 缺 frontmatter/description"
            break
    results.append(_ok("rules.frontmatter") if fm_ok
                   else _bad("rules.frontmatter", fm_detail))

    bridge = root / ".agents" / "rules"
    want = sorted({p.name for p in rules_dir.glob("*.md")} - {"README.md"}) \
        if rules_dir.is_dir() else []
    have = sorted(p.name for p in bridge.glob("*.md")) if bridge.is_dir() else []
    link_ok = have == want and all(
        (bridge / n).is_symlink()
        and os.readlink(bridge / n) == f"../../.claude/rules/{n}" for n in want)
    results.append(_ok("omp.symlinks") if link_ok
                   else _bad("omp.symlinks", f"want={want} have={have}"))

    omp_md = (root / ".omp" / "AGENTS.md")
    omp_body = ("<!-- cadence-managed:omp-context:v1 -->\n"
                "@../.claude/CLAUDE.md\n@../AGENTS.md\n")
    results.append(_ok("omp.agents-md")
                   if omp_md.is_file() and omp_md.read_text(encoding="utf-8") == omp_body
                   else _bad("omp.agents-md", "缺失或非受管内容"))

    agents_md = _read(root, "AGENTS.md") or ""
    n_lines = len(agents_md.splitlines())
    results.append(_ok("agents-md.budget") if n_lines <= 200
                   else _bad("agents-md.budget", f"{n_lines} 行超预算"))
```
