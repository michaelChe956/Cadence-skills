# MCP 宿主车道实施计划（HTTP MCP 五端兼容性观测 · 宿主交互车道）

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 交付 HTTP MCP 五端兼容性观测的宿主交互车道：新增 `eval/hostlane/` 包（五态判定 `verdicts` / curl 全序列前置门 `gate` / PTY 交互驱动 `pty_driver` / 入口编排 `probe`），`night.py` M 组路由分流（不建容器、P/R/D 行为不变），`report_matrix.py` 五态渲染 + M 组分母全豁免，并完成首轮五端实测复现基准签名（codex ⚙×3、其余 ✅ 或有因 ⛔/⊘）与透视快照落档。

**Architecture:** 车道核心是「忠实度四维」的执行链：前置门（真网关真 key 只读、任一失败整轮 ENV_FAIL 零记录）→ 真实 HOME 下五端 PTY 交互会话（禁 headless `-p`；首条消息固化挂载清单，随后 M1-M3 自然语言原话）→ 前后快照差分定位宿主 transcript → 复用五端 adapter（omp 复用 pi）解析 → `classify` 五态判定 → schema 1.0 记录以 `host-<agent>-<ts>-<probe>` 前缀落 docker-night 同池目录（`_dump_run_record` 目录约定）供 `report_matrix` 聚合。判定态全部非门禁：M 组五态均不进端健康度分母。

**Tech Stack:** Python 3.12+ stdlib（`urllib.request`/`json`/`hashlib`/`pathlib`）、pexpect（PTY 交互）、pytest + unittest（测试，经 `uv run --with pytest --with pyyaml --with pexpect` 注入）、既有 eval 资产（`eval/probes/definitions.M_GROUP`、`eval/runner/proc.SESSION_PATTERNS`、`eval/adapters.get_adapter`、`eval/ifmt.IntermediateTrajectory`、`eval/scoring/schema.build_result`）。

**Spec:** `cadence/designs/2026-09-11_方案设计_HTTP_MCP五端兼容性观测体系宿主车道_v1.0.md`（v1.0，唯一需求源：忠实度四维、五态判定、车道设计、验收口径）；`openspec/changes/add-mcp-host-lane/specs/mcp-host-lane/spec.md`（R1-R6 验收对齐；`tasks.md` T0-T8 与本计划任务号一一对应）。

```python
# -*- coding: utf-8 -*-
"""eval/tests/test_hostlane.py —— 宿主车道（HTTP MCP 五端兼容观测）单测。

按任务增量追加测试类（T1 verdicts / T2 gate / T3 pty_driver / T4 probe /
T5 night 路由 / T6 report_matrix）；各类 setUp 内自行 import 被测模块，
保证各任务「先红（模块缺失）后绿」的 TDD 粒度。运行：
  uv run --with pytest --with pyyaml --with pexpect python -m pytest eval/tests/test_hostlane.py -v
"""
import json
import unittest
from pathlib import Path
from unittest import mock


class TestClassify(unittest.TestCase):
    """T1：classify 五态分支 + 按端工具名策略（F1）+ 挂载期失败签名（F2）。"""

    def setUp(self):
        from eval.hostlane import verdicts
        self.v = verdicts

    @staticmethod
    def _call(tool, is_error):
        from eval.ifmt import ToolCall
        return ToolCall(index=0, tool=tool, raw_tool=tool,
                        args_digest="", is_error=is_error)

    def _traj(self, calls=None, final_text=""):
        from eval.ifmt import IntermediateTrajectory
        traj = IntermediateTrajectory(agent="claude", final_text=final_text)
        traj.tool_calls = calls or []
        return traj

    def test_verdicts_vocab(self):
        """ut-hl-vocab：五态词表逐一命名且顺序固定（矩阵渲染依赖）。"""
        self.assertEqual(self.v.VERDICTS,
                         ("PASS", "CALL_FAIL", "NOT_MOUNTED",
                          "SETUP_BLOCKED", "INFRA_FAIL"))

    # ---- F1：按端工具名形态（本机实证）----

    def test_tool_name_strategy_claude_prefix(self):
        """ut-hl-f1-claude：claude=mcp__web_search_prime__* 前缀命中；他端前缀不误配。"""
        self.assertTrue(self.v.tool_matches(
            "claude", "mcp__web_search_prime__web_search_prime", "web-search-prime"))
        self.assertFalse(self.v.tool_matches(
            "claude", "mcp__MiniMax__web_search", "web-search-prime"))

    def test_tool_name_strategy_codex_both_forms(self):
        """ut-hl-f1-codex：codex=原名/下划线化双形态前缀（实证 mcp__codegraph 形，K4）。"""
        self.assertTrue(self.v.tool_matches(
            "codex", "mcp__web_search_prime__search_prime", "web-search-prime"))
        self.assertTrue(self.v.tool_matches(
            "codex", "mcp__web-search-prime__search", "web-search-prime"))
        self.assertFalse(self.v.tool_matches(
            "codex", "mcp__web_reader__read", "web-search-prime"))

    def test_tool_name_strategy_omp_both_forms(self):
        """ut-hl-f1-omp：omp 保大小写——原名/下划线化双形态前缀都试。"""
        self.assertTrue(self.v.tool_matches(
            "omp", "mcp__MiniMax__web_search", "MiniMax"))
        self.assertTrue(self.v.tool_matches(
            "omp", "mcp__web_search_prime__search_prime", "web-search-prime"))

    def test_tool_name_strategy_pi_substring(self):
        """ut-hl-f1-pi：pi 无 mcp__ 前缀——工具名子串匹配（裸名/代理名）。"""
        self.assertTrue(self.v.tool_matches("pi", "MiniMax_web_search", "MiniMax"))
        self.assertTrue(self.v.tool_matches(
            "pi", "web_search_prime", "web-search-prime"))
        self.assertFalse(self.v.tool_matches("pi", "web_reader", "web-search-prime"))
        self.assertFalse(self.v.tool_matches("pi", "Bash", "zread"))

    def test_tool_name_strategy_kimi_registered(self):
        """ut-hl-f1-kimi：kimi 显式登记 prefix（形态假设，T7 首轮实证后回填——K3）。"""
        self.assertEqual(self.v.TOOL_NAME_STRATEGY["kimi"], "prefix")
        self.assertTrue(self.v.tool_matches(
            "kimi", "mcp__web_search_prime__search", "web-search-prime"))

    # ---- 五态分支（三源合成）----

    def test_pass_on_non_error_call(self):
        """ut-hl-pass：非错误目标调用→PASS（claude 前缀形）。"""
        traj = self._traj([self._call("mcp__web_search_prime__search", False)])
        verdict, reason = self.v.classify("claude", "web-search-prime", True, traj)
        self.assertEqual(verdict, "PASS")
        self.assertIn("非错误调用", reason)

    def test_pass_beats_fail_when_both(self):
        """ut-hl-pass-precedence：成功与失败调用并存→PASS（成功即真实使用）。"""
        traj = self._traj([self._call("mcp__web_search_prime__search", True),
                           self._call("mcp__web_search_prime__search", False)])
        verdict, _ = self.v.classify("claude", "web-search-prime", True, traj)
        self.assertEqual(verdict, "PASS")

    def test_call_fail_with_signature(self):
        """ut-hl-callfail：失败调用→CALL_FAIL，reason 携带错误签名摘要。"""
        traj = self._traj([self._call("mcp__web_search_prime__search", True)],
                          final_text="Transport channel closed，重试 3 轮失败")
        verdict, reason = self.v.classify("claude", "web-search-prime", True, traj)
        self.assertEqual(verdict, "CALL_FAIL")
        self.assertIn("channel closed", reason)

    def test_call_fail_on_mount_failure_signature(self):
        """ut-hl-f2-sig：零调用+输出签名命中→CALL_FAIL 挂载期失败，优先于清单。"""
        traj = self._traj()  # 零调用（codex channel closed 时 transcript 无调用）
        verdict, reason = self.v.classify(
            "codex", "web-search-prime", False, traj,
            session_output="… MCP client web-search-prime failed to start …")
        self.assertEqual(verdict, "CALL_FAIL")
        self.assertIn("挂载期失败", reason)

    def test_call_fail_signature_overrides_missing_traj(self):
        """ut-hl-f2-notraj：transcript 缺失但本 server 签名命中仍 CALL_FAIL（R4 ⚙ 可达）。"""
        verdict, _ = self.v.classify(
            "codex", "web-search-prime", False, None,
            session_output="web-search-prime: Transport channel closed")
        self.assertEqual(verdict, "CALL_FAIL")

    def test_mount_failure_signature_is_server_scoped(self):
        """ut-hl-f2-scope：仅他 server 的失败签名不翻本 server 判定（K2 绑定）。"""
        out = "MCP client web-search-prime failed to start（其余正常）"
        verdict, _ = self.v.classify("codex", "zread", False,
                                     self._traj(), session_output=out)
        self.assertEqual(verdict, "NOT_MOUNTED")

    def test_mount_fail_pattern_variants(self):
        """ut-hl-f2-regex：三种签名形态都命中；无关输出不命中。"""
        pat = self.v.MOUNT_FAIL_PATTERN
        self.assertTrue(pat.search("MCP client for server failed to start"))
        self.assertTrue(pat.search("Transport channel closed"))
        self.assertTrue(pat.search("MCP startup failed"))
        self.assertFalse(pat.search("搜索完成，两句话总结"))

    def test_not_mounted_when_absent_from_list(self):
        """ut-hl-notmounted：零调用、无签名、清单不含 server→NOT_MOUNTED。"""
        verdict, reason = self.v.classify(
            "pi", "web-reader", False, self._traj())
        self.assertEqual(verdict, "NOT_MOUNTED")
        self.assertIn("web-reader", reason)

    def test_setup_blocked_missing_traj(self):
        """ut-hl-blocked-traj：挂载、无签名、transcript 缺失→SETUP_BLOCKED。"""
        verdict, reason = self.v.classify("kimi", "zread", True, None)
        self.assertEqual(verdict, "SETUP_BLOCKED")
        self.assertIn("transcript", reason)
```

```python
"""宿主车道五态判定词表与 classify（设计 §3.3，全部非门禁）。

五态语义：
  PASS          transcript 存在非错误的目标 server 调用（真实使用）
  CALL_FAIL     存在失败调用尝试，或零调用但会话输出命中挂载期失败签名
                （挂载后调不动=客户端×网关兼容性问题，如 codex channel closed）
  NOT_MOUNTED   挂载清单无该 server（客户端不挂载此类型=产品事实）
  SETUP_BLOCKED 授权/信任未通过、transcript 缺失（或挂载但零调用证据）
  INFRA_FAIL    进程异常/前置门未过——不在本函数判定，由入口（probe.py）标注

按端工具名形态（2026-09-11 本机实证，评审 F1/K3/K4）：
  claude  mcp__web_search_prime__web_search_prime（server 名 - → _）
  codex   实证 mcp__codegraph 形（无连字符可证，下划线化未证→双形态都试——K4）
  omp     mcp__MiniMax__web_search（server 名保大小写——原名/下划线化双形态都试）
  pi      裸名/代理名（search、MiniMax_web_search、codegraph.search，无 mcp__ 前缀）
          → 工具名子串匹配（server 配置名/下划线化名，大小写不敏感）
  kimi    未实证——显式登记 prefix（形态假设，T7 首轮实证后回填——K3）
"""
import re
from typing import Optional

VERDICTS = ("PASS", "CALL_FAIL", "NOT_MOUNTED", "SETUP_BLOCKED", "INFRA_FAIL")

# 各端目标 server 工具名的匹配策略（F1：mcp__<server>__ 前缀仅 claude 满足）
TOOL_NAME_STRATEGY = {
    "claude": "prefix",       # mcp__<server 下划线化>__*
    "codex": "prefix_both",   # 原名/下划线化双形态（实证 mcp__codegraph 形，K4）
    "omp": "prefix_both",     # 原名/下划线化双形态（server 名保大小写，如 mcp__MiniMax__）
    "pi": "substring",        # 无前缀：工具名含 server 配置名/下划线化名（大小写不敏感）
    "kimi": "prefix",         # 形态假设（未实证）——T7 首轮实证后回填（K3）
}

# 挂载期失败签名（F2：codex 挂载期失败时 transcript 零调用，签名只在会话输出里）
MOUNT_FAIL_PATTERN = re.compile(
    r"MCP client .* failed to start|channel closed|MCP startup failed", re.I)


def _server_forms(server: str) -> tuple:
    """server 名两形态：（原名, 下划线化名），如 (web-search-prime, web_search_prime)。"""
    return server, server.replace("-", "_")


def tool_matches(agent: str, tool: str, server: str) -> bool:
    """按端策略判断 tool 是否为目标 server 的调用（五端策略表显式登记）。"""
    raw, underscored = _server_forms(server)
    strategy = TOOL_NAME_STRATEGY.get(agent, "prefix")
    if strategy == "prefix":
        return f"mcp__{underscored}__" in tool
    if strategy == "prefix_both":
        return f"mcp__{underscored}__" in tool or f"mcp__{raw}__" in tool
    # pi：裸名/代理名无 mcp__ 前缀——子串匹配（大小写不敏感）
    return (underscored.lower() in tool.lower()
            or raw.lower() in tool.lower())


def _mount_failure_line(server: str, session_output: str) -> Optional[str]:
    """在会话输出中找「该 server」的挂载期失败行（K2）。

    同一行需同时命中签名与该 server 原名/下划线名——签名不绑定 server 时，
    单 server 失败会把同端全部探针翻成 CALL_FAIL，误伤真 NOT_MOUNTED。
    """
    raw, underscored = _server_forms(server)
    for line in (session_output or "").splitlines():
        if MOUNT_FAIL_PATTERN.search(line) and (raw in line or underscored in line):
            return line.strip()[:160]
    return None


def classify(agent: str, server: str, mount_listed: bool, traj,
             session_output: str = "") -> tuple:
    """(verdict, reason)——三源合成（Contract 1 评审 F1/F2 修订）：

    ① transcript 调用（按端名称策略）：非错误调用→PASS（成功优先）；
       失败调用→CALL_FAIL（reason 携带 final_text 错误签名摘要）
    ② 零调用时会话输出命中挂载期失败签名且同行绑定该 server（K2）→CALL_FAIL
       （codex 挂载期失败的唯一证据路径，优先于挂载清单）
    ③ 挂载清单不含 server→NOT_MOUNTED；traj 缺失→SETUP_BLOCKED；
       兜底 SETUP_BLOCKED（挂载+有 transcript+零调用记录）。
    """
    calls = ([c for c in traj.tool_calls if tool_matches(agent, c.tool, server)]
             if traj is not None else [])
    ok_calls = [c for c in calls if not c.is_error]
    if ok_calls:
        return "PASS", f"存在非错误调用 ×{len(ok_calls)}（{agent} 端 {server} 工具）"
    if calls:
        sig = (traj.final_text or "").strip().replace("\n", " ")[:120]
        return "CALL_FAIL", f"调用失败 ×{len(calls)}（{server}）签名：{sig or '无 final_text'}"
    fail_line = _mount_failure_line(server, session_output)
    if fail_line:
        return "CALL_FAIL", f"挂载期失败（零调用，会话输出签名）：{fail_line}"
    if not mount_listed:
        return "NOT_MOUNTED", f"挂载清单未列出 {server}（客户端不挂载此类型）"
    if traj is None:
        return "SETUP_BLOCKED", "transcript 缺失（会话文件未定位或解析失败）"
    return ("SETUP_BLOCKED",
            f"挂载清单含 {server} 但 transcript 无该 server 任何调用记录（授权/信任未过或探针未触发）")
```

```python
"""宿主车道网络前置门：三端点 curl 全序列（initialize→session-id→initialized）。

真实网关真实 key（只读自仓库根 .mcp.json）；任一失败由调用方整轮记
ENV_FAIL 且零记录入表——网络层问题不得冒充兼容性结论（设计 §3.2 忠实度四维之网络）。
"""
import json
import urllib.error
import urllib.request
from pathlib import Path

REPO_ROOT = Path(__file__).resolve().parents[2]
MCP_CONFIG = REPO_ROOT / ".mcp.json"
GATE_SERVERS = ("web-search-prime", "web-reader", "zread")

_INIT_BODY = {"jsonrpc": "2.0", "id": 1, "method": "initialize", "params": {
    "protocolVersion": "2024-11-05", "capabilities": {},
    "clientInfo": {"name": "cadence-host-lane", "version": "1.0"}}}
_INITIALIZED_BODY = {"jsonrpc": "2.0", "method": "notifications/initialized"}


def load_server(server: str) -> dict:
    """只读加载仓库根 .mcp.json 的 {url, authorization}；配置缺该 server 抛 KeyError。"""
    cfg = json.loads(MCP_CONFIG.read_text(encoding="utf-8"))
    block = cfg["mcpServers"][server]
    return {"url": block["url"], "authorization": block["headers"]["Authorization"]}


def _post(url, body, authorization, timeout, session_id=None):
    """POST JSON-RPC；返回 (status, headers, body_text)。"""
    headers = {"Content-Type": "application/json",
               "Accept": "application/json, text/event-stream",
               "Authorization": authorization}
    if session_id:
        headers["mcp-session-id"] = session_id
    req = urllib.request.Request(url, data=json.dumps(body).encode("utf-8"),
                                 headers=headers, method="POST")
    with urllib.request.urlopen(req, timeout=timeout) as resp:
        return resp.status, dict(resp.headers), resp.read().decode("utf-8", "replace")


def _body_json(body_text: str):
    """HTTP 200 且含 JSON-RPC 对象（裸 JSON 或 SSE data: 行）才算有效；None=无效。"""
    text = body_text.strip()
    if text.startswith("{"):
        try:
            doc = json.loads(text)
        except ValueError:
            return None
        return doc if isinstance(doc, dict) else None
    for line in text.splitlines():
        line = line.strip()
        if line.startswith("data:"):
            try:
                doc = json.loads(line[5:].strip())
            except ValueError:
                continue
            if isinstance(doc, dict):
                return doc
    return None


def preflight_gate(timeout=20) -> dict:
    """三端点全序列探测：{server: (ok, detail)}——任一失败由调用方整轮 ENV_FAIL。"""
    out = {}
    for server in GATE_SERVERS:
        try:
            info = load_server(server)
        except (OSError, ValueError, KeyError) as exc:
            out[server] = (False, f"配置不可读: {type(exc).__name__}: {exc}")
            continue
        try:
            status, headers, body = _post(info["url"], _INIT_BODY,
                                          info["authorization"], timeout)
            doc = _body_json(body)
            session_id = headers.get("mcp-session-id") or headers.get("Mcp-Session-Id")
            if status != 200 or doc is None:
                out[server] = (False, f"initialize HTTP {status}，"
                                      f"有效 JSON-RPC 响应={'有' if doc else '无'}")
                continue
            if not session_id:
                out[server] = (False, "initialize 200 但响应头缺 mcp-session-id")
                continue
            status2, _, _ = _post(info["url"], _INITIALIZED_BODY,
                                  info["authorization"], timeout,
                                  session_id=session_id)
            if status2 >= 300:
                out[server] = (False, f"notifications/initialized HTTP {status2}")
            else:
                out[server] = (True, f"initialize 200 + session-id + initialized {status2}")
        except urllib.error.HTTPError as exc:
            out[server] = (False, f"HTTP {exc.code}: {exc.reason}")
        except Exception as exc:  # noqa: BLE001 —— 超时/URLError 等网络层异常
            out[server] = (False, f"{type(exc).__name__}: {exc}")
    return out
```

```python
"""宿主 PTY 交互驱动：五端交互式会话（禁 headless -p）、首条消息固化挂载清单、
前后快照差分定位新 transcript（语义参考 eval/runner/proc.py::_pick_new_session）。
"""
import time
from pathlib import Path
from typing import Optional

import pexpect

from eval.runner.proc import SESSION_PATTERNS

# 五端宿主会话目录（HOME 相对 glob）——pi/kimi 沿用 proc.SESSION_PATTERNS；
# claude/codex/omp 与容器版 night.SESSION_GLOBS 同源（HOME 换为宿主真实值）。
HOST_SESSION_GLOBS = {
    "claude": [".claude/projects/**/*.jsonl"],
    "codex": [".codex/sessions/**/*.jsonl"],
    "pi": list(SESSION_PATTERNS["pi"]),
    "kimi": list(SESSION_PATTERNS["kimi"]),
    "omp": [".omp/agent/sessions/**/*.jsonl"],
}

# 交互式启动命令（无 -p/headless——忠实度四维之「交互」）
CLI_COMMANDS = {
    "claude": ["claude"],
    "codex": ["codex"],
    "pi": ["pi"],
    "kimi": ["kimi"],
    "omp": ["omp"],
}

# 会话首条消息：挂载清单指令（忠实度四维之「挂载」——各端自发发现的证据）
MOUNT_LIST_PROMPT = "请逐行列出当前会话可用的全部 MCP 工具名，只要工具名本身，不要解释。"

# 轮次结束判定：TUI 无统一提示符，按「静默 idle 秒」判一轮回复完成（设计 §六 风险缓解）
QUIET_IDLE_S = 20.0


def _scan(home: Path, agent: str) -> dict:
    """按 HOST_SESSION_GLOBS 扫描 home 下候选会话文件，返回 {path: mtime}。"""
    found = {}
    for pattern in HOST_SESSION_GLOBS[agent]:
        for path in home.glob(pattern):
            try:
                found[path] = path.stat().st_mtime
            except OSError:
                continue
    return found


def pick_new_session(before: dict, after: dict) -> Optional[Path]:
    """前后快照差分：本次新建或被追写的最新 session（proc._pick_new_session 语义）。"""
    best, best_mt = None, -1.0
    for path, mtime in after.items():
        if path in before and mtime <= before[path]:
            continue  # 本次调用未动过的旧会话
        if mtime > best_mt:
            best, best_mt = path, mtime
    return best


def _await_quiet(child, limit_s: float, idle_s: float = None) -> str:
    """读到静默 idle_s 或到达 limit_s；返回累计输出（单轮文本捕获）。"""
    if idle_s is None:
        idle_s = QUIET_IDLE_S
    parts = []
    deadline = time.monotonic() + limit_s
    while time.monotonic() < deadline:
        try:
            chunk = child.read_nonblocking(size=4096, timeout=idle_s)
        except (pexpect.TIMEOUT, pexpect.EOF, OSError, ValueError):
            break
        if chunk:
            parts.append(chunk)
    return "".join(parts)


def _exit_child(child) -> None:
    """尽力退出交互 CLI：EOT → /quit → Ctrl-C → 强杀。"""
    for action in (lambda: child.sendcontrol("d"),
                   lambda: child.sendline("/quit"),
                   lambda: child.sendcontrol("c")):
        if not child.isalive():
            return
        try:
            action()
            child.expect(pexpect.EOF, timeout=3)
        except Exception:  # noqa: BLE001 —— 退出尽力而为，最终 close(force)
            continue
    child.close(force=True)


def run_session(agent, messages, cwd, timeout_per_turn=360):
    """起交互 PTY 会话：首条消息=挂载清单指令，随后逐条发送 messages。
    返回 {"transcript_path": str|None, "session_output": 完整 PTY 输出文本}
    （F2：供 classify 挂载期失败签名合成与挂载段切分）；transcript 由前后
    快照差分定位（新会话或被追写的最新文件），定位失败 transcript_path=None
    （调用方按 SETUP_BLOCKED 处理）。进程异常向上抛（入口标 INFRA_FAIL）。
    """
    home = Path.home()
    argv = CLI_COMMANDS[agent]
    before = _scan(home, agent)
    child = pexpect.spawn(argv[0], argv[1:], cwd=str(cwd), encoding="utf-8",
                          timeout=timeout_per_turn, dimensions=(50, 200))
    parts = []
    try:
        parts.append(_await_quiet(child, timeout_per_turn))  # 启动横幅/就绪
        for msg in [MOUNT_LIST_PROMPT] + list(messages):
            child.sendline(msg)
            parts.append(_await_quiet(child, timeout_per_turn))
    finally:
        _exit_child(child)
    tpath = pick_new_session(before, _scan(home, agent))
    return {"transcript_path": str(tpath) if tpath else None,
            "session_output": "".join(parts)}
```

```python
编排（设计 §3.1）：前置门（任一失败=整轮 ENV_FAIL 零记录）→ 真实 HOME 交互会话

transcript → classify 五态 → run_id=host-<agent>-<ts>-<probe> 落
eval/results/docker-night/reports/nightly/<day>/runs/（与 docker 车道同池供矩阵聚合）。

import hashlib
import json
import time
from pathlib import Path

from eval.adapters import get_adapter
from eval.hostlane.gate import preflight_gate
from eval.hostlane.pty_driver import run_session
from eval.hostlane.verdicts import classify

def _dump_run_record(doc: dict) -> Path:
    """run 记录按日落盘——目录约定与 eval/docker/night.py::_dump_run_record 一致
    （宿主车道同池；run_id 前缀 host-<agent>- 与 docker- 区分车道）。"""
    day = time.strftime("%Y-%m-%d")
    runs_dir = RESULT_ROOT / "reports" / "nightly" / day / "runs"
    runs_dir.mkdir(parents=True, exist_ok=True)
    path = runs_dir / f"{doc['run_id']}.json"
    path.write_text(json.dumps(doc, ensure_ascii=False, indent=2),
                    encoding="utf-8")
    return path


def _config_digest(cwd: Path) -> str:
    """项目 .mcp.json 指纹——车道唯一配置触点是只读；前后一致=「配置零改写」证据。"""
    try:
        return hashlib.sha256((cwd / ".mcp.json").read_bytes()).hexdigest()
    except OSError:
        return ""


def _mount_section(output: str, first_probe: str) -> str:
    """挂载清单证据段：首条探针回显之前的部分（挂载清单回复先于任何探针消息）。

    回显不可定位（TUI 重绘等）时退化为全文——最坏把 NOT_MOUNTED 误判为
    SETUP_BLOCKED（可解释注记），不会反向假绿。
    """
    marker = first_probe[:12]
    idx = output.find(marker)
    return output if idx < 0 else output[:idx]


def _mount_listed(section: str, server: str) -> bool:
    """挂载段是否列出该 server（server 名/下划线变体/工具名前缀三式匹配）。"""
    u = server.replace("-", "_")
    return server in section or u in section or f"mcp__{u}__" in section


def _probe_server(probe_id: str) -> str:
    for a in PROBES[probe_id].get("assertions", []):
        if a.get("kind") == "mcp_called":
            return a["server"]
    raise ValueError(f"探针 {probe_id} 无 mcp_called 断言（宿主车道仅服务 M 组）")


def run_user_mcp_probe(agent, probe_ids=("M1", "M2", "M3"), project_cwd=None) -> dict:
    """宿主车道单端一轮：返回 {status, gate, records[, fidelity]}。

    status：OK（逐探针五态记录落盘）/ ENV_FAIL（前置门未过，零记录）/
    INFRA_FAIL（会话进程异常，逐探针 INFRA_FAIL 记录落盘）。
    """
    gate = preflight_gate()
    if not all(ok for ok, _ in gate.values()):
        failed = "；".join(f"{s}: {d}" for s, (ok, d) in gate.items() if not ok)
        print(f"[host-lane] ENV_FAIL：curl 前置门未过，本轮零记录——{failed}")
        return {"status": "ENV_FAIL", "gate": gate, "records": []}

    prompts = [PROBES[p]["prompt_variants"][0] for p in probe_ids]
    cwd = Path(project_cwd) if project_cwd else REPO_ROOT
    sha_before = _config_digest(cwd)
    ts = int(time.time())
    t0 = time.monotonic()
    try:
        sess = run_session(agent, prompts, cwd)
    except Exception as exc:  # noqa: BLE001 —— 进程异常由入口标 INFRA_FAIL
        reason = f"进程异常: {type(exc).__name__}: {exc}"[:200]
        print(f"[host-lane] INFRA_FAIL {agent}: {reason}")
        records = [result_schema.build_result(
            run_id=f"host-{agent}-{ts}-{p}", agent=agent, probe_id=p,
            verdict="INFRA_FAIL", fail_reason=reason, duration_s=0.0,
            details={"lane": "host", "gate": gate, "infra": reason})
            for p in probe_ids]
        for doc in records:
            _dump_run_record(doc)
        return {"status": "INFRA_FAIL", "gate": gate, "records": records}

    duration = round(time.monotonic() - t0, 1)
    output, tpath = sess["session_output"], sess["transcript_path"]
    traj = None
    if tpath:
        try:
            adapter = get_adapter("pi" if agent == "omp" else agent)
            traj = adapter.parse_file(Path(tpath))
        except Exception as exc:  # noqa: BLE001 —— 解析失败→SETUP_BLOCKED 可解释
            print(f"[host-lane] transcript 解析失败（判 SETUP_BLOCKED）: {exc}")
    sha_after = _config_digest(cwd)
    section = _mount_section(output, prompts[0]) if prompts else output
    records = []
    for p in probe_ids:
        server = _probe_server(p)
        mount_listed = _mount_listed(section, server)
        verdict, reason = classify(agent, server, mount_listed, traj,
                                   session_output=output)
        doc = result_schema.build_result(
            run_id=f"host-{agent}-{ts}-{p}", agent=agent, probe_id=p,
            verdict=verdict, fail_reason=reason, duration_s=duration,
            transcript_path=str(tpath or ""),
            details={"lane": "host", "mount_listed": mount_listed,
                     "gate": gate, "config_zero_write": sha_before == sha_after,
                     "mount_section": section[:2000],
                     "session_output": output[:2000],
                     "final_text": (traj.final_text if traj else "")[:2000]})
        _dump_run_record(doc)
        records.append(doc)
    fidelity = {"config_zero_write": sha_before == sha_after,
                "mount_section": section[:500],
                "natural_language_prompts": True,
                "gate": gate}
    return {"status": "OK", "gate": gate, "records": records, "fidelity": fidelity}
```

```python
# M 组（宿主车道）五态符号与格渲染——观测非门禁，与 docker 车道 ✅/⚠️/❌/⊘ 并行
_M_VERDICT_SYMBOL = {"PASS": "✅", "CALL_FAIL": "⚙", "NOT_MOUNTED": "⊘",
                     "SETUP_BLOCKED": "⛔", "INFRA_FAIL": "❌"}
_M_VERDICT_LABEL = {"PASS": "全过", "CALL_FAIL": "调用失败", "NOT_MOUNTED": "未挂载",
                    "SETUP_BLOCKED": "受阻", "INFRA_FAIL": "进程异常"}  # K5：ENV_FAIL 零记录永不入表


def _m_cell(runs: list) -> str:
    """M 组格：单态=符号+标签+n/n；混合轮次并列各态计数（⚙1·⊘1）。"""
    n = len(runs)
    by_v = {}
    for v, _ in runs:
        by_v[v] = by_v.get(v, 0) + 1
    if len(by_v) == 1:
        v = next(iter(by_v))
        return f"{_M_VERDICT_SYMBOL.get(v, '？')} {_M_VERDICT_LABEL.get(v, v)} {n}/{n}"
    return "⚠️ " + "·".join(
        f"{_M_VERDICT_SYMBOL.get(v, '？')}{c}" for v, c in sorted(by_v.items()))
```
